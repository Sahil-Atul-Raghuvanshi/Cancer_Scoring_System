# This app has no requirements file of its own to install, and that is deliberate.
#
# `start.bat` runs it with the demo backend's virtual environment
# (`../tissue_scoring_demo/backend/.venv`), which already carries
# every package below. A second environment would mean a second 200 MB CPU torch
# download to run the same code against the same weights.
#
# The list is here so that what the app actually depends on is written down, and so it
# can be installed elsewhere if it ever needs to be:
#
#     pip install -r backend/requirements.txt
#
# Note what is NOT here: `nnunetv2`, `dynamic_network_architectures`, `batchgenerators`,
# `acvl_utils`, `SimpleITK`. nnU-Net is needed only to turn `plans.json` into a
# `torch.nn.Module`, and `backend/bracs_app/unet.py` does that in 200 lines of plain
# torch instead - which is why this app installs on a Python 3.13 CPU box where several
# of those packages have no wheel. See that module's docstring for what makes the
# substitution safe.

torch>=2.4,<3          # CPU build is sufficient; the teacher runs at ~0.6 s per 512 patch
numpy>=2.0
pillow>=10.0           # `Image.fromarray` on uint16 gives mode I;16 without the
                       # deprecated `mode=` argument, which Pillow 13 removes
scipy>=1.11            # gaussian_filter, for the sliding-window importance map

# Whole-slide reading, for the six-slide comparison screen only. NOT openslide-python:
# that needs a system OpenSlide library which is not on this machine, and tiffslide reads
# Aperio SVS through `tifffile` with no binary dependency at all. The demo backend's
# `app.ingestion.slide_reader` - the one place in the workspace that opens a slide - is
# already written against it, so this is recording an existing dependency rather than
# adding one.
tiffslide>=2.2
fastapi>=0.110
uvicorn>=0.27
pydantic>=2.6
python-multipart>=0.0.9  # only for the upload form on screen 4; FastAPI raises a clear
                         # RuntimeError naming it if a multipart route is hit without it

# Development only.
pytest>=8.0
httpx>=0.27            # fastapi.testclient
